 321  def _save_upload(file, custom_name=""):
 322      if not file or not file.filename:
 323          return jsonify({"error": "请选择文件"}), 400
 324      filename = secure_filename(custom_name.strip() or file.filename)
 325      if not filename:
 326          return jsonify({"error": "文件名无效"}), 400
 327      extension = filename.rsplit(".", 1)[-1].casefold() if "." in filename else ""
 328      if extension not in ALLOWED_EXTENSIONS:
 329          security_logger.warning("upload_rejected username=%s ip=%s filename=%s reason=extension", session.get("username"), client_ip(), filename)
 330          return jsonify({"error": "不支持的文件类型"}), 400
 331      if file.content_length and file.content_length > app.config["MAX_CONTENT_LENGTH"]:
 332          return jsonify({"error": "文件大小超过限制"}), 413
 333      signatures = FILE_SIGNATURES.get(extension)
 334      if signatures:
 335          header = file.stream.read(16)
 336          file.stream.seek(0)
 337          if not any(header.startswith(signature) for signature in signatures):
 338              security_logger.warning("upload_rejected username=%s ip=%s filename=%s reason=file_signature", session.get("username"), client_ip(), filename)
 339              return jsonify({"error": "文件内容与扩展名不匹配"}), 400
 340      user_dir = os.path.join(app.config["UPLOAD_FOLDER"], session["username"])
 341      os.makedirs(user_dir, exist_ok=True)
 342      file.save(os.path.join(user_dir, filename))
 343      security_logger.info("upload_success username=%s ip=%s filename=%s", session.get("username"), client_ip(), filename)
 344      return jsonify({"success": True, "file": filename}), 200
